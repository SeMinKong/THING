# ============================================================================
# thing_control_web / frontend 환경변수 템플릿
# ----------------------------------------------------------------------------
# 사용법
#   cd web/frontend
#   cp env.txt .env.local        # .env.local 은 .gitignore 대상이다
#
# Vite 는 `.env` 와 `.env.local` 을 모두 읽고 `.env.local` 이 우선한다.
# 저장소의 `.env` 에는 로컬 개발 기본값만 두고, 장비별 실제 주소는
# `.env.local` 에 적어 커밋하지 않는다.
#
# 주의: Vite 는 `VITE_` 로 시작하는 변수만 클라이언트에 노출한다.
# 여기 적는 값은 모두 브라우저 번들에 그대로 들어가므로 비밀값을 두지 않는다.
#
# 근거: FR-28 "frontend 는 ... `WS_URL` 과 `MJPEG_URL` 외부 설정으로 Jetson Web
#      Bridge 와 MJPEG 에 연결한다", FR-20 영상 표시
# ============================================================================


# ── VITE_WS_URL ────────────────────────────────────────────────────────────
#
# 6.4절이 서버→클라이언트 endpoint 를 /ws/robot-state 로 고정한다.
# 그 endpoint 를 제공하는 것은 Jetson 의 thing_web_bridge ROS 2 노드다.
# 브라우저가 여기에 직접 붙는다. 중계 서버는 없다.
#
# 비워두면 현재 호스트를 기준으로 ws://<host>/ws/robot-state 를 만들고
# vite dev 프록시가 VITE_DEV_WS_TARGET(기본 ws://localhost:8000)으로 넘긴다.
# Laptop 에서 Vite 를 띄우고 Jetson 에 붙는 구성에서는 아래를 지정한다.
#
#   VITE_WS_URL=ws://192.168.0.10:8000/ws/robot-state
#
VITE_WS_URL=


# ── VITE_MJPEG_STREAM_URL (필수) ───────────────────────────────────────────
#
# landmark overlay 가 합성된 MJPEG 스트림. thing_vision(Jetson)이 제공한다.
# 비워두면 화면에 "스트림 주소가 아직 설정되지 않았습니다" 가 표시된다.
#
#   VITE_MJPEG_STREAM_URL=http://192.168.0.10:8080/stream/overlay
#
VITE_MJPEG_STREAM_URL=


# ── VITE_MJPEG_RAW_STREAM_URL (선택) ───────────────────────────────────────
#
# 원본 RGB 스트림. 별도 엔드포인트가 있을 때만 지정한다.
# 비워두면 "원본 영상" 선택 버튼이 표시되지 않고 overlay 만 보여준다.
#
#   VITE_MJPEG_RAW_STREAM_URL=http://192.168.0.10:8080/stream/raw
#
VITE_MJPEG_RAW_STREAM_URL=


# ── VITE_DEV_WS_TARGET (선택, 개발 전용) ───────────────────────────────────
#
# VITE_WS_URL 을 비워둔 채로 개발할 때 vite dev 서버가 /ws/robot-state 를
# 넘길 대상. 기본값은 ws://localhost:8000 이다.
#
#   VITE_DEV_WS_TARGET=ws://192.168.0.10:8000
#
VITE_DEV_WS_TARGET=
